# Demonstrates proper scheme structure

# Example 1: Complete scheme with all fields
murder_scheme = {
	skill = intrigue
	desc = murder_scheme_desc
	icon = "gfx/interface/icons/schemes/murder.dds"
	category = hostile
	cooldown = { years = 5 }

	is_secret = yes
	base_secrecy = 30

	target_type = character
	is_valid_target = {
		is_alive = yes
	}

	valid_agent = {
		age >= 16
		intrigue >= 8
		NOT = { has_relation_rival = scope:target }
	}

	agent_join_chance = {
		base = 0

		modifier = {
			add = 25
			has_relation_friend = scope:owner
		}

		modifier = {
			add = 50
			has_trait = ambitious
		}
	}

	base_progress_goal = 180

	on_ready = {
		scope:owner = {
			trigger_event = murder.0001
		}
	}

	on_invalidated = {
		scope:owner = {
			trigger_event = murder.invalidated
		}
	}

	on_monthly = {
		scope:owner = {
			add_intrigue_lifestyle_xp = 10
		}
	}
}

# Example 3: Scheme with agent_join_chance modifiers and secrecy
spy_network_scheme = {
	skill = intrigue
	desc = spy_network_scheme_desc
	icon = "gfx/interface/icons/schemes/spy_network.dds"
	category = hostile
	cooldown = { years = 3 }

	is_secret = yes
	base_secrecy = 50

	target_type = character
	is_valid_target = {
		is_ruler = yes
		is_alive = yes
		NOT = { this = scope:owner }
	}

	valid_agent = {
		age >= 16
		is_adult = yes
		NOT = { is_close_family_of = scope:target }
	}

	agent_join_chance = {
		base = 10

		modifier = {
			add = 20
			has_relation_friend = scope:owner
		}

		modifier = {
			add = 30
			has_trait = deceitful
		}

		modifier = {
			add = -20
			has_relation_friend = scope:target
		}

		modifier = {
			add = 15
			opinion = {
				target = scope:owner
				value >= 40
			}
		}

		modifier = {
			add = -30
			has_trait = honest
		}
	}

	on_agent_join = {
		scope:agent = {
			add_opinion = {
				target = scope:owner
				modifier = conspirator_opinion
			}
		}
	}

	on_agent_leave = {
		scope:agent = {
			add_opinion = {
				target = scope:owner
				modifier = abandoned_plot_opinion
			}
		}
	}

	base_progress_goal = 200

	on_ready = {
		scope:owner = {
			trigger_event = spy_network.0001
		}
	}

	on_invalidated = {
		scope:owner = {
			trigger_event = spy_network.invalidated
		}
	}

	on_exposed = {
		scope:owner = {
			add_prestige = -100
		}
		scope:target = {
			trigger_event = spy_network.exposed
		}
	}
}

# Example 4: Scheme with complex on_ready logic involving multiple outcomes
abduct_scheme = {
	skill = intrigue
	desc = abduct_scheme_desc
	icon = "gfx/interface/icons/schemes/abduct.dds"
	category = hostile

	is_secret = yes
	base_secrecy = 20

	target_type = character
	is_valid_target = {
		is_alive = yes
		is_landed = yes
		NOT = { this = scope:owner }
	}

	valid_agent = {
		age >= 16
		prowess >= 6
	}

	agent_join_chance = {
		base = 5

		modifier = {
			add = 25
			has_trait = sadistic
		}

		modifier = {
			add = 15
			has_relation_rival = scope:target
		}
	}

	base_progress_goal = 240

	on_ready = {
		# Multiple outcome paths based on agent count and target state
		if = {
			limit = {
				scheme_number_of_agents >= 3
				scope:target = {
					is_at_war = yes
				}
			}
			# Best outcome: target distracted by war, many agents
			scope:owner = {
				trigger_event = abduct.success_easy
			}
		}
		else_if = {
			limit = {
				scheme_number_of_agents >= 1
			}
			# Normal outcome: enough agents for the job
			scope:owner = {
				trigger_event = abduct.success_normal
			}
		}
		else = {
			# Difficult outcome: no agents, owner goes alone
			scope:owner = {
				trigger_event = abduct.success_risky
			}
		}
	}

	on_invalidated = {
		scope:owner = {
			trigger_event = abduct.invalidated
		}
	}

	on_monthly = {
		# Secrecy degrades faster with more agents
		if = {
			limit = {
				scheme_number_of_agents >= 3
			}
			add_scheme_modifier = {
				type = large_conspiracy_secrecy_penalty
			}
		}
		scope:owner = {
			add_intrigue_lifestyle_xp = 5
		}
	}
}
