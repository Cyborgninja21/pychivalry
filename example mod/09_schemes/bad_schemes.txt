# Bad Schemes Examples
# Demonstrates SCHEME-001 to SCHEME-003

# ERROR SCHEME-001: Missing skill field
bad_scheme_001 = {
	# ERROR SCHEME-001: No skill field
	# Should have: skill = intrigue (or diplomacy, stewardship, etc.)

	desc = bad_scheme_001_desc
	category = hostile

	target_type = character
	is_valid_target = {
		is_alive = yes
	}

	base_progress_goal = 180

	on_ready = {
		scope:owner = {
			add_gold = 100
		}
	}
}

# ERROR SCHEME-002: No effects
bad_scheme_002 = {
	skill = intrigue
	desc = bad_scheme_002_desc
	category = hostile

	target_type = character
	is_valid_target = {
		is_alive = yes
	}

	base_progress_goal = 180

	# ERROR SCHEME-002: No on_ready, on_invalidated, or any effect blocks
	# Scheme does nothing when ready
}

# ERROR SCHEME-003: Uses agents but no valid_agent conditions
bad_scheme_003 = {
	skill = intrigue
	desc = bad_scheme_003_desc
	category = hostile

	target_type = character
	is_valid_target = {
		is_alive = yes
	}

	# ERROR SCHEME-003: Has agent_join_chance but no valid_agent block
	agent_join_chance = {
		base = 25

		modifier = {
			add = 50
			has_trait = ambitious
		}
	}

	base_progress_goal = 180

	on_ready = {
		scope:owner = {
			add_prestige = 100
		}
	}
}

	base_progress_goal = 180

	# ERROR SCHEME-002: Empty on_ready
	on_ready = {
	}

# ERROR SCHEME-003: Agent logic without valid_agent
bad_scheme_agents_no_validation = {
	skill = intrigue
	desc = bad_scheme_agents_no_validation_desc

	target_type = character

	# ERROR SCHEME-003: No valid_agent block to define who can be an agent
	agent_join_chance = {
		base = 25

		modifier = {
			add = 25
			has_trait = ambitious
		}

		modifier = {
			add = 50
			has_relation_friend = scope:owner
		}
	}

	base_progress_goal = 180

	on_ready = {
		add_prestige = 100
	}
}
