namespace = scope_timing_bad

# ===================================================================
# THE GOLDEN RULE (violated in all examples below):
# Event Evaluation Order:
# 1. trigger = { }    <- Evaluated FIRST
# 2. desc = { }       <- Evaluated SECOND
# 3. immediate = { }  <- Runs THIRD (scopes created here)
# 4. Portraits        <- Displayed FOURTH
# 5. Options          <- Rendered FIFTH
#
# VIOLATION: Scopes created in 'immediate' are NOT available in 'trigger' or 'desc'!

# -------------------------------------------------------------------
# ERROR CK3550: Scope in trigger from immediate
# -------------------------------------------------------------------
# WHAT THIS TESTS: Detection of scopes used before creation
# SHOULD FAIL: Yes - CK3550 error
# WHY IT'S BAD: trigger runs BEFORE immediate, scope doesn't exist yet
# HOW TO FIX: Save scope in trigger, not immediate
# SEVERITY: ERROR - will crash or cause unpredictable behavior
# -------------------------------------------------------------------
scope_timing_bad.0001 = {
	type = character_event
	title = scope_timing_bad.0001.t
	desc = scope_timing_bad.0001.desc

	trigger = {
		is_adult = yes
		# ERROR CK3550: scope:my_liege doesn't exist yet!
		# It's created in immediate (which runs AFTER trigger)
		scope:my_liege = {
			is_alive = yes
		}
	}

	immediate = {
		save_scope_as = actor
		# Saving scope here, but trigger already ran
		liege = {
			save_scope_as = my_liege
		}
	}

	option = {
		name = scope_timing_bad.0001.a
		add_gold = 100
	}
}

# -------------------------------------------------------------------
# Scope in desc from immediate (WARNING)
# -------------------------------------------------------------------
# WHAT THIS TESTS: Detection of scopes in desc from immediate
# SHOULD FAIL: Yes - CK3551 warning
# WHY IT'S BAD: desc evaluates BEFORE immediate, scope doesn't exist
# HOW TO FIX: Save scope in trigger if needed in desc
# SEVERITY: WARNING - desc will be broken/empty
# -------------------------------------------------------------------
scope_timing_bad.0002 = {
	type = character_event
	title = scope_timing_bad.0002.t
# desc block references scope created in immediate
	desc = {
		triggered_desc = {
			trigger = {
				# scope:rival doesn't exist yet! (created in immediate)
				scope:rival = {
					is_alive = yes
				}
			}
			desc = scope_timing_bad.0002.desc_rival
		}
		desc = scope_timing_bad.0002.desc_default
	}

	trigger = {
		is_adult = yes
	}

	immediate = {
		# Too late - desc already evaluated
		random_courtier = {
			limit = {
				has_relation_rival = root
			}
			save_scope_as = rival
		}
	}

	option = {
		name = scope_timing_bad.0002.a
		add_gold = 100
	}
}
