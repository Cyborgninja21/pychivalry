# Good On-Actions Examples
# Demonstrates proper on_action structure and usage

namespace = on_actions_good

# Example 1: Simple on_action with events list
on_birth = {
	events = {
		on_actions_good.0001
		on_actions_good.0002
	}
}

on_actions_good.0001 = {
	type = character_event
	title = on_actions_good.0001.t
	desc = on_actions_good.0001.desc

	left_portrait = root

	immediate = {
		add_gold = 100
	}

	option = {
		name = on_actions_good.0001.a
		add_prestige = 50
	}
}

on_actions_good.0002 = {
	type = character_event
	title = on_actions_good.0002.t
	desc = on_actions_good.0002.desc

	left_portrait = root

	immediate = {
		add_piety = 25
	}

	option = {
		name = on_actions_good.0002.a
		add_gold = 50
	}
}

# Example 3: On_action with effects
on_yearly_pulse = {
	effect = {
		every_living_character = {
			limit = {
				is_adult = yes
				is_ruler = yes
			}
			add_gold = 10
		}
	}
}

# Example 5: On_action with chance_to_happen
on_five_year_pulse = {
	events = {
		on_actions_good.0007
	}
}

on_actions_good.0007 = {
	type = character_event
	title = on_actions_good.0007.t
	desc = on_actions_good.0007.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	weight_multiplier = {
		base = 1

		modifier = {
			factor = 2
			has_trait = ambitious
		}
	}

	immediate = {
		add_gold = 500
	}

	option = {
		name = on_actions_good.0007.a
		add_prestige = 250
	}
}
