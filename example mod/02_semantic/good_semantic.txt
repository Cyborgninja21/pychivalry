# -------------------------------------------------------------------
# Example 1: Valid triggers
# -------------------------------------------------------------------
# WHAT THIS TESTS: Correct trigger keywords and boolean logic
# SHOULD PASS: Yes - all triggers are valid CK3 keywords
# DEMONSTRATES: Proper use of triggers in event trigger block
# -------------------------------------------------------------------
semantic_good.0001 = {
	type = character_event
	title = semantic_good.0001.t
	desc = semantic_good.0001.desc

	trigger = {
		# Valid character triggers
		is_adult = yes
		is_ruler = yes
		is_alive = yes
		age >= 16
		gold >= 100
		prestige >= 50

		# Valid scope chain triggers
		liege = {
			is_alive = yes
		}

		# Valid boolean logic
		OR = {
			has_trait = brave
			has_trait = ambitious
		}

		NOT = {
			has_trait = craven
		}
	}

	option = {
		name = semantic_good.0001.a
		# Valid effects
		add_gold = 100
		add_prestige = 50
		add_piety = 25
	}
}

# -------------------------------------------------------------------
# Example 4: Valid list iterations
# -------------------------------------------------------------------
# WHAT THIS TESTS: Proper use of any_, every_, and random_ iterators
# SHOULD PASS: Yes - correct context for each iterator type
# DEMONSTRATES: any_ in triggers (checking), every_ and random_ in effects (modifying)
# -------------------------------------------------------------------
semantic_good.0004 = {
	type = character_event
	title = semantic_good.0004.t
	desc = semantic_good.0004.desc

	trigger = {
		# Valid any_ iterator (trigger context)
		any_vassal = {
			is_adult = yes
			has_trait = ambitious
		}

		# Valid any_ with complex conditions
		any_courtier = {
			age >= 16
			NOT = {
				has_trait = lazy
			}
		}
	}

	immediate = {
		# Valid every_ iterator (effect context)
		every_vassal = {
			limit = {
				is_adult = yes
			}
			add_gold = 10
		}

		# Valid random_ iterator
		random_courtier = {
			limit = {
				age >= 16
			}
			add_prestige = 50
		}
	}

	option = {
		name = semantic_good.0004.a
		add_prestige = 100
	}
}

# -------------------------------------------------------------------
# Example 5: Valid saved scopes
# -------------------------------------------------------------------
# WHAT THIS TESTS: Proper save_scope_as and scope: reference usage
# SHOULD PASS: Yes - scopes saved before use
# DEMONSTRATES: Correct workflow for saving and referencing scopes
# -------------------------------------------------------------------
semantic_good.0005 = {
	type = character_event
	title = semantic_good.0005.t
	desc = semantic_good.0005.desc

	trigger = {
		is_adult = yes
	}

	immediate = {
		# Save scopes for later use
		save_scope_as = actor

		liege = {
			save_scope_as = my_liege
		}

		# Use saved scopes
		scope:my_liege = {
			add_opinion = {
				target = scope:actor
				modifier = grateful_opinion
			}
		}
	}

	option = {
		name = semantic_good.0005.a
		# Use saved scope in effects
		scope:my_liege = {
			add_prestige = 50
		}
	}
}

# -------------------------------------------------------------------
# Example 6: Valid exists checks before operations
# -------------------------------------------------------------------
# WHAT THIS TESTS: Proper exists checks to prevent null reference crashes
# SHOULD PASS: Yes - exists checked before is_alive
# DEMONSTRATES: Best practice for safe scope navigation
# NOTE: This prevents CK5137 errors
# -------------------------------------------------------------------
semantic_good.0006 = {
	type = character_event
	title = semantic_good.0006.t
	desc = semantic_good.0006.desc

	trigger = {
		# Good practice: check exists before is_alive
		father = {
			exists = yes
			is_alive = yes
		}
	}

	immediate = {
		# Good practice: exists check in limit
		if = {
			limit = {
				liege = {
					exists = yes
				}
			}
			liege = {
				add_gold = 100
			}
		}
	}

	option = {
		name = semantic_good.0006.a
		add_prestige = 50
	}
}
