namespace = scripted_bad

# -------------------------------------------------------------------
# ERROR CK3950: Undefined scripted effect
# -------------------------------------------------------------------
# WHAT THIS TESTS: Detection of calls to scripted effects that don't exist
# SHOULD FAIL: Yes - CK3950 error
# WHY IT'S BAD: The scripted effect name is not defined anywhere
# HOW TO FIX: Define the effect in common/scripted_effects/ or fix the name
# -------------------------------------------------------------------
scripted_bad.0001 = {
	type = character_event
	title = scripted_bad.0001.t
	desc = scripted_bad.0001.desc

	trigger = {
		is_adult = yes
	}

	option = {
		name = scripted_bad.0001.a
		grant_epic_loot_effect = yes # ERROR CK3950: Undefined scripted effect: "grant_epic_loot_effect"
	}

	option = {
		name = scripted_bad.0001.b
		summon_dragon_effect = yes # ERROR CK3950: Undefined scripted effect: "summon_dragon_effect"
	}
}

# -------------------------------------------------------------------
# ERROR CK3956: Recursive scripted block — block calls itself
# -------------------------------------------------------------------
# WHAT THIS TESTS: Detection of scripted blocks that call themselves
# SHOULD FAIL: Yes - CK3956 warning
# WHY IT'S BAD: Infinite recursion will crash the game or hang
# HOW TO FIX: Remove the self-reference or refactor into separate effects
# -------------------------------------------------------------------
recursive_reward_effect = {
	add_gold = 50
	recursive_reward_effect = yes # ERROR CK3956: Scripted block 'recursive_reward_effect' calls itself recursively
}

recursive_check_trigger = {
	is_adult = yes
	recursive_check_trigger = yes # ERROR CK3956: Scripted block 'recursive_check_trigger' calls itself recursively
}
