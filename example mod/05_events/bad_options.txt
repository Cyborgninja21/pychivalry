# Bad Options Examples
# Demonstrates CK3450-CK3461 (option validation) and CK3610-CK3614 (AI chance validation)

# ERROR: CK3420
# ERROR: CK3872
# ERROR: CONV-004
# ERROR: EVENT-007
# ERROR: LOC-001
namespace = events_bad_options

# ERROR CK3450: Option missing name
events_bad_options.0001 = {
	type = character_event
	title = events_bad_options.0001.t
	desc = events_bad_options.0001.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		# ERROR CK3450: Missing 'name = ...' field
		add_gold = 100
	}

	option = {
		name = events_bad_options.0001.b
		add_prestige = 50
	}
}

# Redundant fallback option
events_bad_options.0004 = {
	type = character_event
	title = events_bad_options.0004.t
	desc = events_bad_options.0004.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_options.0004.a
		add_gold = 100
	}

	option = {
		name = events_bad_options.0004.b
# This is the fallback option, 'always = yes' is redundant
		trigger = {
			always = yes
		}
		add_prestige = 50
	}
}

# Multiple exclusive options may conflict
events_bad_options.0005 = {
	type = character_event
	title = events_bad_options.0005.t
	desc = events_bad_options.0005.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_options.0005.a
# Both options have mutually exclusive triggers
		trigger = {
			has_trait = brave
		}
		exclusive = yes
		add_gold = 100
	}

	option = {
		name = events_bad_options.0005.b
		trigger = {
			has_trait = craven
		}
		exclusive = yes # ERROR CK3455: Conflict if neither trait exists
		add_prestige = 50
	}
}

# All options have triggers (no fallback)
events_bad_options.0009 = {
	type = character_event
	title = events_bad_options.0009.t
	desc = events_bad_options.0009.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

# All options have triggers - no guaranteed fallback
	option = {
		name = events_bad_options.0009.a
		trigger = {
			gold >= 100
		}
		add_gold = 100
	}

	option = {
		name = events_bad_options.0009.b
		trigger = {
			prestige >= 50
		}
		add_prestige = 50
	}

	option = {
		name = events_bad_options.0009.c
		trigger = {
			piety >= 25
		}
		add_piety = 25
	}

	# If player has no gold, prestige, or piety, event won't display!
}
