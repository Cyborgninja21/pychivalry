# Bad Event Structure Examples
# Demonstrates CK3760-CK3769 - Event structure validation errors

namespace = events_bad_structure

# ERROR CK3760: Event missing type declaration
events_bad_structure.0001 = {
	# ERROR CK3760: No 'type = character_event' or similar
	title = events_bad_structure.0001.t
	desc = events_bad_structure.0001.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_structure.0001.a
		add_gold = 100
	}
}

# ERROR CK3761: Invalid event type
events_bad_structure.0002 = {
	type = super_awesome_event # ERROR CK3761: Not a valid event type
	title = events_bad_structure.0002.t
	desc = events_bad_structure.0002.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_structure.0002.a
		add_gold = 100
	}
}

# ERROR CK3762: Hidden event with options
events_bad_structure.0003 = {
	type = character_event
	hidden = yes # Hidden events shouldn't have options

	trigger = {
		is_adult = yes
	}

	immediate = {
		add_gold = 100
	}

	# ERROR CK3762: Hidden events don't display options
	option = {
		name = events_bad_structure.0003.a
		add_prestige = 50
	}

	option = {
		name = events_bad_structure.0003.b
		add_piety = 25
	}
}

# ERROR CK3763: Event with no option blocks
events_bad_structure.0004 = {
	type = character_event
	title = events_bad_structure.0004.t
	desc = events_bad_structure.0004.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	immediate = {
		add_gold = 100
	}

	# ERROR CK3763: No options - player has no way to dismiss event
}

# ERROR CK3764: Non-hidden event missing desc
events_bad_structure.0005 = {
	type = character_event
	title = events_bad_structure.0005.t
	# ERROR CK3764: Missing desc field (required for non-hidden events)

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_structure.0005.a
		add_gold = 100
	}
}

# ERROR CK3766: Multiple after blocks
events_bad_structure.0006 = {
	type = character_event
	title = events_bad_structure.0006.t
	desc = events_bad_structure.0006.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_structure.0006.a
		add_gold = 100
	}

	# ERROR CK3766: Only the first after block will execute
	after = {
		trigger_event = {
			id = events_bad_structure.0007
			days = 30
		}
	}

	# Second after block - will be ignored
	after = {
		add_prestige = 50
	}
}

events_bad_structure.0007 = {
	type = character_event
	title = events_bad_structure.0007.t
	desc = events_bad_structure.0007.desc

	left_portrait = root

	option = {
		name = events_bad_structure.0007.a
		add_gold = 50
	}
}

# ERROR CK3767: Empty event block
events_bad_structure.0008 = {
	# ERROR CK3767: Event has no content at all
}

# ERROR CK3768: Multiple immediate blocks
events_bad_structure.0009 = {
	type = character_event
	title = events_bad_structure.0009.t
	desc = events_bad_structure.0009.desc

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	# ERROR CK3768: Only one immediate block allowed
	immediate = {
		add_gold = 100
	}

	# Second immediate block - only first executes
	immediate = {
		add_prestige = 50
	}

	option = {
		name = events_bad_structure.0009.a
		add_piety = 25
	}
}

# ERROR CK3769: Character event has no portraits
events_bad_structure.0010 = {
	type = character_event
	title = events_bad_structure.0010.t
	desc = events_bad_structure.0010.desc

	# ERROR CK3769: No left_portrait, right_portrait, or any portraits defined
	# Visual events should show character portraits

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_structure.0010.a
		add_gold = 100
	}
}

	desc = events_bad_structure.0013.desc # ERROR CK3762: Desc ignored in hidden event

	}
