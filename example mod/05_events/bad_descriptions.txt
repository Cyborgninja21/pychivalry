# Bad Descriptions Examples
# Demonstrates CK3440-CK3446 - Description block validation errors

namespace = events_bad_desc

# ERROR CK3440: triggered_desc missing trigger
events_bad_desc.0001 = {
	type = character_event
	title = events_bad_desc.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				# ERROR CK3440: Missing trigger block
				desc = events_bad_desc.0001.desc_brave
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = events_bad_desc.0001.desc_default
			}
		}
	}

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_desc.0001.a
		add_gold = 100
	}
}

# ERROR CK3446: Excessive desc nesting (>3 levels)
events_bad_desc.0007 = {
	type = character_event
	title = events_bad_desc.0007.t
	desc = {
		# ERROR CK3446: Too deeply nested (4+ levels)
		first_valid = {
			triggered_desc = {
				trigger = {
					has_trait = brave
				}
				desc = {
					first_valid = {
						triggered_desc = {
							trigger = {
								has_trait = ambitious
							}
							desc = {
								first_valid = {
									triggered_desc = {
										trigger = {
											has_trait = greedy
										}
