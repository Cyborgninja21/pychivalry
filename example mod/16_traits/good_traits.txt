# Good Traits Examples
# Demonstrates correct vanilla trait references in events

namespace = traits_good

# Example 1: Basic trait checks and additions
traits_good.0001 = {
	type = character_event
	title = traits_good.0001.t
	desc = traits_good.0001.desc

	left_portrait = root

	trigger = {
		is_adult = yes
		has_trait = brave
	}

	immediate = {
		save_scope_as = actor
	}

	option = {
		name = traits_good.0001.a
		add_trait = ambitious
		add_prestige = 50
	}

	option = {
		name = traits_good.0001.b
		add_trait = humble
		add_piety = 50
	}
}

# Example 2: Personality trait removal and opposites
traits_good.0002 = {
	type = character_event
	title = traits_good.0002.t
	desc = traits_good.0002.desc

	left_portrait = root

	trigger = {
		has_trait = craven
		is_adult = yes
	}

	immediate = {
		save_scope_as = actor
	}

	option = {
		name = traits_good.0002.a
		# Overcome cowardice
		remove_trait = craven
		add_trait = brave
		add_prestige = 100
	}

	option = {
		name = traits_good.0002.b
		# Remain cautious
		add_gold = 50
	}
}

# Example 5: Complex trait interactions with multiple checks
traits_good.0005 = {
	type = character_event
	title = traits_good.0005.t
	desc = traits_good.0005.desc

	left_portrait = root

	trigger = {
		is_adult = yes
		is_ruler = yes
		NOT = { has_trait = incapable }
	}

	immediate = {
		save_scope_as = actor
	}

	option = {
		name = traits_good.0005.a
		trigger = {
			has_trait = arrogant
			has_trait = ambitious
		}
		add_prestige = 200
		add_gold = 100
	}

	option = {
		name = traits_good.0005.b
		trigger = {
			has_trait = gregarious
		}
		add_prestige = 100
		every_vassal = {
			limit = { is_adult = yes }
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 10
			}
		}
	}

	option = {
		name = traits_good.0005.c
		trigger = {
			OR = {
				has_trait = depressed_1
				has_trait = stressed_1
			}
		}
		remove_trait = depressed_1
		remove_trait = stressed_1
		add_piety = 50
	}
}
