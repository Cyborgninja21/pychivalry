	title = authoring_unrest.0001.t

	option = {

		ai_chance = { base = 50 }
		scope:unrest_county = { change_county_control = 20 }
		add_gold = -50
	}

	option = {
		name = authoring_unrest.0001.b
		ai_chance = { base = 50 }
		scope:unrest_county = { add_popular_opinion = 10 }
		add_prestige = -25
	}
}
