namespace = authoring_envoy

# An envoy arrives at court.
authoring_envoy.0001 = {
	type = character_event
	title = authoring_envoy.0001.t
	desc = authoring_envoy.0001.desc

	trigger = { authoring_envoy_host_trigger = yes }

	option = {
		name = authoring_envoy.0001.a
		random_courtier = {
			limit = { is_adult = yes }
			save_scope_as = envoy
		}
		trigger_event = {
			id = authoring_envoy.0002
			days = 5
		}
	}
}
