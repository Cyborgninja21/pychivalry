namespace = harvest

# The court gathers for the festival.
harvest.0004 = {
	type = character_event
	title = harvest.0004.t
	desc = harvest.0004.desc

	trigger = {
		is_ruler = yes
		harvest_can_host_trigger = yes
	}

	immediate = {
		every_courtier = {
			limit = { is_adult = yes }
			add_stress = -5
		}
	}

	option = {
		name = harvest.0004.a
		add_prestige = 15
		harvest_reward_effect = yes
	}
	option = {
		name = harvest.0004.b
		add_gold = -10
		add_stress = -5
	}
}
