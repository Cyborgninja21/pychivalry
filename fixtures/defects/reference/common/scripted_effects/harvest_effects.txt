# Effects run by the festival's events and decisions.
harvest_feast_effect = {
	if = {
		limit = { is_adult = yes }
		add_prestige = 20
	}
	else_if = {
		limit = { is_ai = yes }
		add_gold = -5
	}
	random_list = {
		60 = {
			add_gold = 25
		}
		40 = {
			add_piety = 15
		}
	}
	add_piety = 10
}

# The host takes the tithe.
harvest_tithe_effect = {
	ROOT = {
		add_gold = 25
	}
}

# Gives alms to the poor of the capital.
harvest_alms_giving_effect = {
	add_gold = -15
	add_piety = 15
	capital_county = {
		add_county_modifier = {
			modifier = harvest_granary_modifier
			days = 180
		}
	}
}
