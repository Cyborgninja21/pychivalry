namespace = harvest

# The faith blesses the fields.
harvest.0010 = {
	scope = faith
	hidden = yes
	title = harvest.0010.t
	desc = harvest.0010.desc

	trigger = {
		fervor <= 90
	}

	immediate = {
		change_fervor = 3
	}

	option = {
		name = harvest.0010.a
	}
}

# The faithful give thanks for the harvest.
harvest.0011 = {
	type = character_event
	title = harvest.0011.t
	desc = harvest.0011.desc

	trigger = {
		faith = { fervor >= 30 }
	}

	option = {
		name = harvest.0011.a
		add_piety = 20
		faith = {
			change_fervor = 2
		}
	}
	option = {
		name = harvest.0011.b
		add_prestige = 10
	}
}
