# ERROR STORY-026: first_valid missing unconditional fallback
bad_story_026 = {
	on_owner_death = {
		scope:story = {
			end_story = yes
		}
	}

	effect_group = {
		months = 1

		first_valid = {
			triggered_effect = {
				trigger = {
					has_trait = brave
				}
				effect = {
					add_gold = 100
				}
			}

			triggered_effect = {
				trigger = {
					has_trait = craven
				}
				effect = {
					add_prestige = 50
				}
			}

			# ERROR STORY-026: No always = yes fallback
			# What if character has neither trait?
		}
	}
}
