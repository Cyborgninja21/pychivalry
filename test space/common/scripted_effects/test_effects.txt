# -------------------------------------------------------------------
# Example 1: Feast rewards effect with scope iteration
# -------------------------------------------------------------------
# WHAT THIS TESTS: Effect with stress_impact and every_vassal
# SHOULD PASS: Yes - valid effects and scope usage
# DEMONSTRATES: Complex effect with opinion modifiers
# REFERENCED BY: common/on_action/test_on_actions.txt,
#                common/scripted_guis/test_sgui.txt
# -------------------------------------------------------------------
apply_feast_rewards_effect = {
	add_prestige = 200
	add_piety = 50
	stress_impact = {
		gregarious = medium_stress_loss
		shy = medium_stress_gain
	}
	every_vassal = {
		limit = {
			opinion = {
				target = root
				value >= 0
			}
		}
		add_opinion = {
			target = root
			modifier = feast_opinion
			opinion = 10
		}
	}
}
