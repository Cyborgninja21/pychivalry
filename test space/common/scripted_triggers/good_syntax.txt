	type = character_event
	title = syntax_good.0002.t
	desc = syntax_good.0002.desc

	trigger = {
		age >= 16
		OR = {
			is_ruler = yes
			AND = {
				is_landed = yes
				is_adult = yes
			}
		}
	}

	option = {
		name = syntax_good.0002.a

			limit = {
				gold >= 100
			}

		}

			add_piety = 25
		}
	}
}
