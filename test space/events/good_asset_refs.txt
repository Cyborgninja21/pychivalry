# Good Asset References Examples
# Demonstrates proper asset file references

namespace = assets_good

# Example 1: Valid graphics file reference
assets_good.0001 = {
	type = character_event
	title = assets_good.0001.t
	desc = assets_good.0001.desc

	# Valid icon reference to existing file
	picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

	left_portrait = root

	trigger = {
		is_adult = yes
	}

	option = {
		name = assets_good.0001.a
		add_gold = 100
	}
}
