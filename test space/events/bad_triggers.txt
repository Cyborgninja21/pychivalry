# -------------------------------------------------------------------
# ERROR CK5142: Character comparison with = instead of 'this ='
# -------------------------------------------------------------------
# WHAT THIS TESTS: Detection of incorrect character comparison syntax
# SHOULD FAIL: Yes - CK5142 error (2 instances)
# WHY IT'S BAD: 'liege = root' is invalid, use 'liege = { this = root }'
# HOW TO FIX: Use 'this = root' inside a scope block
# -------------------------------------------------------------------

	title = semantic_bad_triggers.0010.t
	desc = semantic_bad_triggers.0010.desc

	trigger = {
		# ERROR CK5142: Use 'this = root' not '= root' for character comparison
		liege = root # Wrong syntax for character comparison

		# ERROR CK5142: Should use 'this = scope:actor'

			save_temporary_scope_as = vassal

		}
	}

	option = {
		name = semantic_bad_triggers.0010.a
		add_prestige = 50
	}
}
