namespace = test_space

# -------------------------------------------------------------------
# Example 1: Complete character event with all standard elements
# -------------------------------------------------------------------
# WHAT THIS TESTS: Basic event structure with trigger, immediate, options
# SHOULD PASS: Yes - proper structure, valid triggers/effects
# DEMONSTRATES: Minimal correct event with event chain
# -------------------------------------------------------------------
test_space.0001 = {
	type = character_event
	title = test_space.0001.t
	desc = test_space.0001.desc
	theme = diplomacy

	left_portrait = root

	trigger = {
		is_alive = yes
		age >= 16
	}

	immediate = {
		save_scope_as = feast_host
		add_prestige = 100
	}

	option = {
		name = test_space.0001.a
		add_gold = 50
		trigger_event = {
			id = test_space.0002
			days = 30
		}
	}

	option = {
		name = test_space.0001.b
		add_piety = 25
	}
}

# -------------------------------------------------------------------
# Example 2: Follow-up event in chain
