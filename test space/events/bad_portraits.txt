# Bad Portraits Examples
# Demonstrates CK3420-CK3422, CK3430 - Portrait and theme validation errors

namespace = events_bad_portraits

# ERROR CK3420: Invalid portrait position
events_bad_portraits.0001 = {
	type = character_event
	title = events_bad_portraits.0001.t
	desc = events_bad_portraits.0001.desc

	# ERROR CK3420: Invalid portrait position names
	top_portrait = root # Should be: left_portrait, right_portrait, center_portrait, etc.
	bottom_portrait = liege
	middle_portrait = father

	trigger = {
		is_adult = yes
	}

	option = {
		name = events_bad_portraits.0001.a
		add_gold = 100
	}
}

# ERROR CK3421: Portrait missing character field

# ERROR CK3422: Invalid animation
events_bad_portraits.0003 = {
	type = character_event
	title = events_bad_portraits.0003.t
	desc = events_bad_portraits.0003.desc

	left_portrait = {
		character = root
		animation = super_dance # ERROR CK3422: Not a valid animation name
	}

	right_portrait = {
		character = liege
		animation = flying_dragon # ERROR CK3422: Invalid animation
	}

	trigger = {
		is_adult = yes
		exists = liege
	}

	option = {
		name = events_bad_portraits.0003.a
		add_gold = 100
	}
}
